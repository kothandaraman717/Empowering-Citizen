    <script>
        // State management
        let currentStep = 0;
        let selectedLang = 'en-IN';
        let isListening = false;
        let recognition = null;
        let userProfile = { age: null, gender: null, category: null, income: null, state: null };

        // Database of Schemes
        const schemesData = [

            },

                maxAge: 60
            },
            {
                id: "pm_awas",
                title: "PM Awas Yojana (PMAY-Gramin)",
                ministry: "Ministry of Rural Development",
                benefit: "Financial assistance up to ₹1.2 Lakh for building a pucca home",
                docs: ["Aadhaar Card", "Job Card (MGNREGA)", "Bank Account Details", "Land Record"],
                category: "all",
                maxIncome: 200000,
                minAge: 18,
                maxAge: 80
            }
        ];

        // Questionnaire Config
        const questions = [
            {
                key: "age",
                text: "What is your age?",
                hint: "Type your age in numbers (e.g., 35)",
                icon: "user",
                type: "number",
                options: ["25", "35", "45", "60"]
            },
            {
                key: "gender",
                text: "Select your gender",
                hint: "Choose or say your gender",
                icon: "users",
                type: "select",
                options: [
                    { label: "Male", value: "male" },
                    { label: "Female", value: "female" },
                    { label: "Other", value: "other" }
                ]
            },
            {
                key: "category",
                text: "What is your primary occupation?",
                hint: "Select the option that best describes your work",
                icon: "briefcase",
                type: "select",
                options: [
                    { label: "Farmer / Agriculture", value: "farmer" },
                    { label: "Small Business / Self-Employed", value: "business" },
                    { label: "Daily Wage Worker / Laborer", value: "laborer" },
                    { label: "Student / Other", value: "other" }
                ]
            },
            {
                key: "income",
                text: "What is your estimated annual family income?",
                hint: "Select your annual income bracket",
                icon: "indian-rupee",
                type: "select",
                options: [
                    { label: "Below ₹1.5 Lakh", value: 150000 },
                    { label: "₹1.5 Lakh - ₹2.5 Lakh", value: 250000 },
                    { label: "₹2.5 Lakh - ₹5 Lakh", value: 500000 },
                    { label: "Above ₹5 Lakh", value: 800000 }
                ]
            },
            {
                key: "state",
                text: "Which state do you reside in?",
                hint: "Select your home state",
                icon: "map-pin",
                type: "select",
                options: [
                    { label: "Uttar Pradesh", value: "UP" },
                    { label: "Madhya Pradesh", value: "MP" },
                    { label: "Tamil Nadu", value: "TN" },
                    { label: "Maharashtra", value: "MH" }
                ]
            }
        ];

        // Speech Recognition Setup
        if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            recognition = new SpeechRecognition();
            recognition.continuous = false;
            recognition.interimResults = false;

            recognition.onresult = function(event) {
                const transcript = event.results[0][0].transcript.toLowerCase();
                document.getElementById('transcriptionBox').innerText = `Heard: "${transcript}"`;
                document.getElementById('transcriptionBox').classList.remove('hidden');
                processVoiceAnswer(transcript);
                stopVoice();
            };

            recognition.onerror = function() {
                document.getElementById('voiceStatus').innerText = "Could not hear clearly. Try again or click below.";
                stopVoice();
            };

            recognition.onend = function() {
                stopVoice();
            };
        }

        function toggleVoiceInput() {
            if (!recognition) {
                alert("Speech recognition is not supported in this browser. Please use Chrome or Edge.");
                return;
            }
            if (isListening) {
                stopVoice();
            } else {
                startVoice();
            }
        }

        function startVoice() {
            isListening = true;
            recognition.lang = selectedLang;
            recognition.start();
            document.getElementById('micBtn').classList.add('pulse-mic', 'bg-red-600');
            document.getElementById('voiceStatus').innerText = "Listening... Speak now";
            speakText(questions[currentStep].text);
        }

        function stopVoice() {
            isListening = false;
            document.getElementById('micBtn').classList.remove('pulse-mic', 'bg-red-600');
            document.getElementById('voiceStatus').innerText = "Click microphone to speak...";
        }

        function speakText(text) {
            if ('speechSynthesis' in window) {
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = selectedLang;
                window.speechSynthesis.speak(utterance);
            }
        }

        function changeLanguage(lang) {
            selectedLang = lang;
            speakText("Language changed");
        }

        function renderQuestion() {
            const q = questions[currentStep];
            document.getElementById('currentStepNum').innerText = currentStep + 1;
            document.getElementById('progressBar').style.width = `${((currentStep + 1) / 5) * 100}%`;
            document.getElementById('progressPercent').innerText = `${((currentStep + 1) / 5) * 100}%`;
            
            document.getElementById('questionText').innerText = q.text;
            document.getElementById('questionHint').innerText = q.hint;
            document.getElementById('transcriptionBox').classList.add('hidden');

            const container = document.getElementById('optionButtons');
            container.innerHTML = '';

            if (q.type === 'number') {
                const input = document.createElement('input');
                input.type = 'number';
                input.id = 'numberInput';
                input.placeholder = 'e.g. 35';
                input.className = 'col-span-2 text-center text-lg font-semibold border-2 border-orange-200 rounded-xl py-3 focus:outline-none focus:border-orange-500';
                input.value = userProfile[q.key] || '';
                container.appendChild(input);
            } else {
                q.options.forEach(opt => {
                    const btn = document.createElement('button');
                    btn.type = 'button';
                    btn.className = `w-full py-3 px-4 rounded-xl text-sm font-semibold border transition text-left flex justify-between items-center ${userProfile[q.key] === opt.value ? 'bg-orange-600 text-white border-orange-600 shadow-md' : 'bg-slate-50 border-slate-200 text-slate-700 hover:bg-orange-50 hover:border-orange-200'}`;
                    btn.innerText = opt.label;
                    btn.onclick = () => {
                        userProfile[q.key] = opt.value;
                        renderQuestion();
                    };
                    container.appendChild(btn);
                });
            }

            document.getElementById('prevBtn').style.visibility = currentStep === 0 ? 'hidden' : 'visible';
            document.getElementById('nextBtn').innerText = currentStep === questions.length - 1 ? 'See Results' : 'Next';
            lucide.createIcons();
        }

        function processVoiceAnswer(transcript) {
            const q = questions[currentStep];
            if (q.type === 'number') {
                const matchedNum = transcript.match(/\d+/);
                if (matchedNum) {
                    userProfile[q.key] = parseInt(matchedNum[0], 10);
                    renderQuestion();
                }
            } else {
                q.options.forEach(opt => {
                    if (transcript.includes(opt.label.toLowerCase()) || transcript.includes(String(opt.value).toLowerCase())) {
                        userProfile[q.key] = opt.value;
                        renderQuestion();
                    }
                });
            }
        }

        function nextQuestion() {
            const q = questions[currentStep];
            if (q.type === 'number') {
                const numVal = document.getElementById('numberInput').value;
                if (numVal) userProfile[q.key] = parseInt(numVal, 10);
            }

            if (!userProfile[q.key]) {
                alert("Please select or enter an answer before proceeding.");
                return;
            }

            if (currentStep < questions.length - 1) {
                currentStep++;
                renderQuestion();
            } else {
                evaluateEligibility();
            }
        }

        function prevQuestion() {
            if (currentStep > 0) {
                currentStep--;
                renderQuestion();
            }
        }

        function evaluateEligibility() {
            document.getElementById('questionnaireCard').classList.add('hidden');
            document.getElementById('quizHeader').classList.add('hidden');
            document.getElementById('resultsSection').classList.remove('hidden');

            const grid = document.getElementById('schemesGrid');
            grid.innerHTML = '';

            const matchedSchemes = schemesData.filter(s => {
                const ageMatch = userProfile.age >= s.minAge && userProfile.age <= s.maxAge;
                const catMatch = s.category === 'all' || s.category === userProfile.category;
                const incomeMatch = userProfile.income <= s.maxIncome;
                const genderMatch = !s.gender || s.gender === userProfile.gender;
                return ageMatch && catMatch && incomeMatch && genderMatch;
            });

            if (matchedSchemes.length === 0) {
                grid.innerHTML = `<p class="col-span-2 text-center text-slate-500 py-6">No direct scheme matches found based on current criteria. Try adjusting income or occupation details.</p>`;
                return;
            }

            matchedSchemes.forEach(scheme => {
                const card = document.createElement('div');
                card.className = "border border-orange-100 bg-orange-50/30 rounded-xl p-5 shadow-sm hover:shadow-md transition flex flex-col justify-between";
                card.innerHTML = `
                    <div>
                        <div class="flex justify-between items-start gap-2 mb-2">
                            <h3 class="font-bold text-slate-900 text-base">${scheme.title}</h3>
                            <span class="bg-emerald-100 text-emerald-800 text-[10px] font-bold px-2 py-0.5 rounded-full">Eligible</span>
                        </div>
                        <p class="text-xs text-orange-700 font-semibold mb-3">${scheme.ministry}</p>
                        <p class="text-xs text-slate-600 mb-4 bg-white p-3 rounded-lg border border-slate-100">${scheme.benefit}</p>
                        <div class="mb-4">
                            <h4 class="text-xs font-bold text-slate-700 mb-1">Required Documents:</h4>
                            <ul class="text-xs text-slate-600 list-disc list-inside space-y-0.5">
                                ${scheme.docs.map(doc => `<li>${doc}</li>`).join('')}
                            </ul>
                        </div>
                    </div>
                    <button onclick="alert('Proceeding to portal for ${scheme.title}')" class="w-full bg-orange-600 hover:bg-orange-700 text-white text-xs font-semibold py-2 rounded-lg transition shadow-sm">
                        Apply Now
                    </button>
                `;
                grid.appendChild(card);
            });
            lucide.createIcons();
        }

        function resetQuestionnaire() {
            currentStep = 0;
            userProfile = { age: null, gender: null, category: null, income: null, state: null };
            document.getElementById('resultsSection').classList.add('hidden');
            document.getElementById('questionnaireCard').classList.remove('hidden');
            document.getElementById('quizHeader').classList.remove('hidden');
            renderQuestion();
        }

        function toggleHighContrast() {
            document.getElementById('appBody').classList.toggle('high-contrast');
        }

        // Initialize on page load
        window.onload = function() {
            renderQuestion();
            lucide.createIcons();
        };
    </script>
